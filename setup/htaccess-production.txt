# Reguly bezpieczenstwa dla hostingu wspoldzielonego.
#
# Lokalnie te same zasady wchodza przez setup/apache-security.conf montowany do
# konfiguracji Apache. Na hostingu wspoldzielonym nie ma dostepu do konfiguracji
# serwera, wiec musza pojechac w .htaccess.
#
# Uzycie: wklej ponizsze reguly do .htaccess w katalogu glownym WordPressa,
# PONIZEJ bloku "# BEGIN WordPress ... # END WordPress". WordPress nadpisuje
# swoj blok przy kazdym zapisie permalinkow i skasowalby to, co stoi w srodku.
#
# Po wgraniu sprawdz, ze /wp-content/debug.log zwraca 403 albo 404.

# Serwer nie musi ogłaszać swojej wersji.
ServerSignature Off

<IfModule mod_headers.c>
    Header always unset X-Powered-By
</IfModule>

# Zrzuty bazy, logi i kopie plikow nie sa tresciami publicznymi. Trafiaja tu
# przez przypadek, przy recznym zgrywaniu albo edytorze zostawiajacym .old.
<FilesMatch "(?i)\.(log|sql|bak|old|dist)$">
    Require all denied
</FilesMatch>

# Pliki kropkowe: .env, .git, .htpasswd i reszta tego, co nigdy nie ma wyjsc
# poza serwer.
<FilesMatch "^\.">
    Require all denied
</FilesMatch>

# xmlrpc.php jest wylaczony w motywie po stronie PHP, ale odciecie go na poziomie
# serwera oszczedza cykle przy probach logowania metoda slownikowa.
<Files xmlrpc.php>
    Require all denied
</Files>

# Wykonywanie PHP w katalogu uploadow. Nawet jesli ktos wgra tam plik, nie
# uruchomi sie. Klasyczna sciezka wejscia po przejeciu konta redaktora.
# Ten blok wklej do osobnego .htaccess w wp-content/uploads/, nie do glownego.
#
# <FilesMatch "\.(?i:php|phtml|php[0-9])$">
#     Require all denied
# </FilesMatch>

# ---------------------------------------------------------------------------
# HSTS. ETAP POCZATKOWY JEST USTAWIANY PRZEZ MOTYW W inc/security.php.
# ---------------------------------------------------------------------------
#
# Naglowek mowi przegladarce: przez najblizsze max-age sekund nie probuj tej
# domeny po HTTP w ogole. To jest zobowiazanie nieodwracalne po stronie
# przegladarki. Nie ma zdalnego wylacznika: skrocenie max-age dziala dopiero
# przy kolejnej wizycie, a przegladarka, ktora zapamietala rok, przez rok nie
# wejdzie na strone, jesli certyfikat wygasnie albo przekierowanie sie zepsuje.
#
# Dlatego kolejnosc jest taka, a nie odwrotna:
#
# 1. Certyfikat dziala, https://winnicanowizny.pl otwiera sie bez ostrzezenia,
#    a http:// przekierowuje na https:// z kodem 301. Sprawdz oba warianty,
#    z www i bez.
# 2. Odkomentuj blok ponizej z max-age=300 (piec minut). Blad kosztuje wtedy
#    piec minut, nie rok.
# 3. Poczekaj tydzien normalnego dzialania strony. Sprawdz, czy odnowienie
#    certyfikatu przez hosting przeszlo automatycznie.
# 4. Podnies max-age do 31536000 (rok) i dodaj includeSubDomains, jesli zadna
#    poddomena nie ma pracowac po HTTP.
#
# Preload zostawiamy poza gra. Wpis na liste preload jest wpisany na stale w
# przegladarke i usuwa sie go miesiacami.
#
# Nie ustawiaj tutaj rownolegle max-age=300: motyw wysyla go juz dla HTTPS na
# produkcji. Po okresie obserwacji zwiekszaj wartosc w inc/security.php, zeby
# naglowek mial jedno zrodlo prawdy.
#
# Docelowo, po kroku 4, przenies te wartosc do inc/security.php:
# <IfModule mod_headers.c>
#     Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
# </IfModule>
