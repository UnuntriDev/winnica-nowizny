# Reguly bezpieczenstwa dla hostingu wspoldzielonego.
#
# Lokalnie te same zasady wchodza przez setup/apache-security.conf montowany do
# konfiguracji Apache. Na hostingu wspoldzielonym nie ma dostepu do konfiguracji
# serwera, wiec musza pojechac w .htaccess.
#
# Uzycie: wklej ponizsze reguly do .htaccess w katalogu glownym WordPressa,
# PONIZEJ bloku "# BEGIN WordPress ... # END WordPress". WordPress nadpisuje
# swoj blok przy kazdym zapisie permalinkow i skasowalby to, co stoi w srodku.
#
# Po wgraniu sprawdz, ze /wp-content/debug.log zwraca 403 albo 404.

# Serwer nie musi ogłaszać swojej wersji.
ServerSignature Off

<IfModule mod_headers.c>
    Header always unset X-Powered-By
</IfModule>

# Zasoby z assets/dist maja hash tresci w nazwie nadawany przez Vite. Nowa
# wersja pliku dostaje nowy URL, wiec mozna je bezpiecznie trzymac rok w cache.
# Regula nie obejmuje obrazow i plikow motywu bez hasha, ktore moga zostac
# podmienione pod ta sama nazwa.
<IfModule mod_setenvif.c>
    SetEnvIf Request_URI "^/wp-content/themes/winnica-nowizny/assets/dist/" winnica_dist_asset=1
</IfModule>

<IfModule mod_headers.c>
    Header set Cache-Control "public, max-age=31536000, immutable" env=winnica_dist_asset
</IfModule>

# Zrzuty bazy, logi i kopie plikow nie sa tresciami publicznymi. Trafiaja tu
# przez przypadek, przy recznym zgrywaniu albo edytorze zostawiajacym .old.
<FilesMatch "(?i)\.(log|sql|bak|old|dist)$">
    Require all denied
</FilesMatch>

# Pliki kropkowe: .env, .git, .htpasswd i reszta tego, co nigdy nie ma wyjsc
# poza serwer.
<FilesMatch "^\.">
    Require all denied
</FilesMatch>

# xmlrpc.php jest wylaczony w motywie po stronie PHP, ale odciecie go na poziomie
# serwera oszczedza cykle przy probach logowania metoda slownikowa.
<Files xmlrpc.php>
    Require all denied
</Files>

# Wykonywanie PHP w katalogu uploadow. Nawet jesli ktos wgra tam plik, nie
# uruchomi sie. Klasyczna sciezka wejscia po przejeciu konta redaktora.
# Ten blok wklej do osobnego .htaccess w wp-content/uploads/, nie do glownego.
#
# <FilesMatch "\.(?i:php|phtml|php[0-9])$">
#     Require all denied
# </FilesMatch>

# HSTS jest ustawiany na poziomie hostingu, aby obejmowal takze odpowiedzi
# serwera, ktore omijaja WordPressa. Wartosc 86400 jest bezpiecznym etapem
# posrednim. Po okresie stabilnej pracy mozna zwiekszac ja stopniowo. Nie
# dodawaj includeSubDomains ani preload bez osobnej weryfikacji subdomen.
<IfModule mod_headers.c>
    Header always set Strict-Transport-Security "max-age=86400"
    Header always set Content-Security-Policy-Report-Only "default-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'self'; object-src 'none'; script-src 'self' 'unsafe-inline'; style-src 'self' 'unsafe-inline'; img-src 'self' data: https:; font-src 'self' data:; connect-src 'self'; frame-src https://www.google.com https://maps.google.com; media-src 'self'; manifest-src 'self'"
</IfModule>
